<script>
let stripe, cardElement;

async function initStripe() {
  if (stripe) return;
  try {
    const cfg = await fetch(window.location.origin + '/api/config').then(r => r.json());
    if (!cfg.stripePublicKey) throw new Error('Stripe not configured');
    stripe = Stripe(cfg.stripePublicKey);
    const elements = stripe.elements({
      fonts: [{ cssSrc: 'https://fonts.googleapis.com/css2?family=Space+Mono&display=swap' }],
    });
    cardElement = elements.create('card', {
      style: {
        base: {
          color: '#f5f0e8',
          fontFamily: '"Space Mono", monospace',
          fontSize: '13px',
          '::placeholder': { color: 'rgba(245,240,232,0.4)' },
        },
        invalid: { color: '#e07070' },
      },
    });
    cardElement.mount('#card-element');
  } catch (err) {
    console.error('Stripe init error:', err.message);
  }
}

document.addEventListener('DOMContentLoaded', initStripe);

async function handleCall(e) {
  e.preventDefault();
  const form    = document.getElementById('call-form');
  const btn     = form.querySelector('button[type="submit"]');
  const msg     = document.getElementById('call-msg');
  const errDiv  = document.getElementById('card-errors');
  const firstName   = document.getElementById('f-name').value.trim();
  const phoneNumber = document.getElementById('f-phone').value.trim();

  btn.disabled     = true;
  btn.textContent  = 'Authorizing...';
  msg.style.display    = 'none';
  errDiv.style.display = 'none';

  try {
    const piRes  = await fetch(window.location.origin + '/api/create-payment-intent', {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify({ firstName, phoneNumber, character: 'CHARACTER_SLUG' }),
    });
    const piData = await piRes.json();
    if (!piRes.ok) throw new Error(piData.error || 'Payment setup failed.');

    const { error: stripeError, paymentIntent } = await stripe.confirmCardPayment(
      piData.clientSecret,
      { payment_method: { card: cardElement, billing_details: { name: firstName } } }
    );
    if (stripeError) {
      errDiv.textContent   = stripeError.message;
      errDiv.style.display = 'block';
      btn.disabled    = false;
      btn.textContent = 'Call Me Now →';
      return;
    }

    btn.textContent = 'Connecting...';
    const callRes  = await fetch(window.location.origin + '/api/start-call-basic', {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify({
        firstName,
        phoneNumber,
        character:        'CHARACTER_SLUG',
        paymentIntentId:  paymentIntent.id,
        stripeCustomerId: piData.stripeCustomerId,
      }),
    });
    const callData = await callRes.json();
    if (callRes.ok) {
      msg.textContent   = 'Your call is connecting — pick up within 30 seconds.';
      msg.style.color   = '#6dbf8a';
      msg.style.display = 'block';
      btn.textContent   = 'Call Placed ✓';
    } else {
      throw new Error(callData.error || 'Something went wrong.');
    }
  } catch (err) {
    msg.textContent   = err.message;
    msg.style.color   = '#e07070';
    msg.style.display = 'block';
    btn.disabled      = false;
    btn.textContent   = 'Call Me Now →';
  }
}
